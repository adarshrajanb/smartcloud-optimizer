def clamp(val: float, low: float = 0.0, high: float = 100.0) -> float:
    return max(low, min(high, val))


def usd_to_inr(value: float, rate: float = 83.5) -> float:
    return value * rate


def fetch_azure_vm_prices(region: str = "eastus", currency: str = "USD") -> List[Dict[str, Any]]:
    params = {
        "$filter": (
            "serviceName eq 'Virtual Machines' "
            f"and armRegionName eq '{region}' "
            "and priceType eq 'Consumption'"
        ),
        "currencyCode": currency,
    }

    try:
        response = requests.get(AZURE_PRICES_URL, params=params, timeout=25)
        response.raise_for_status()
        payload = response.json()
        return payload.get("Items", [])
    except Exception:
        return []


def normalize_azure_prices(region: str = "eastus") -> List[Dict[str, Any]]:
    items = fetch_azure_vm_prices(region=region, currency="USD")
    normalized: List[Dict[str, Any]] = []

    for item in items:
        sku = item.get("armSkuName", "")
        if not sku.startswith("Standard_"):
            continue
        if not any(token in sku for token in ["D", "B", "E", "F"]):
            continue
        if "LowPriority" in sku or "Promo" in sku:
            continue

        retail_price = item.get("retailPrice")
        if retail_price is None:
            continue

        try:
            retail_float = float(retail_price)
        except (TypeError, ValueError):
            continue

        normalized.append(
            {
                "sku": sku,
                "retail_price_usd": retail_float,
                "currency": item.get("currencyCode", "USD"),
                "arm_region": item.get("armRegionName"),
                "meter_name": item.get("meterName"),
                "unit_of_measure": item.get("unitOfMeasure"),
                "product_name": item.get("productName"),
                "inr_per_hour": round(usd_to_inr(retail_float, 83.5), 4),
                "inr_per_month": round(usd_to_inr(retail_float * 730, 83.5), 2),
                "source": "Azure Retail Prices API",
            }
        )

    return normalized[:30]


def tier_by_size(vcpu: int, ram_gb: int) -> dict:
    best = VM_TIERS[0]
    for tier in VM_TIERS:
        if tier["vcpu"] >= vcpu and tier["ram_gb"] >= ram_gb:
            best = tier
            break
    return best


def downsize_tier(current_tier: dict) -> dict:
    idx = next(i for i, t in enumerate(VM_TIERS) if t["id"] == current_tier["id"])
    return VM_TIERS[max(0, idx - 1)]


def upsize_tier(current_tier: dict) -> dict:
    idx = next(i for i, t in enumerate(VM_TIERS) if t["id"] == current_tier["id"])
    return VM_TIERS[min(len(VM_TIERS) - 1, idx + 1)]


def generate_vm_resource(region: str, idx: int, env: str, team: str) -> Dict[str, Any]:
    base_vcpu = {"core-backend": 8, "analytics": 16, "frontend": 4, "qa": 6}[team]
    base_ram = {"core-backend": 16, "analytics": 32, "frontend": 8, "qa": 12}[team]

    tier = tier_by_size(base_vcpu, base_ram)
    if env == "prod":
        tier = tier_by_size(base_vcpu + 4, base_ram + 8)
    if env == "dev":
        tier = tier_by_size(max(2, base_vcpu - 2), max(4, base_ram - 4))

    return {
        "id": f"vm-{region.lower()}-{idx}",
        "name": f"{team}-{env}-{region.lower()}-{idx}",
        "type": "vm",
        "region": region,
        "environment": env,
        "team": team,
        "vcpu": tier["vcpu"],
        "ram_gb": tier["ram_gb"],
        "tier": tier["id"],
        "monthly_cost_inr": tier["monthly_inr"],
        "hourly_cost_inr": tier["hourly_inr"],
        "status": "active",
        "created_at": "2026-01-01T00:00:00Z",
    }


def generate_block_volume(region: str, idx: int, env: str, team: str) -> Dict[str, Any]:
    return {
        "id": f"vol-{region.lower()}-{idx}",
        "name": f"{team}-{env}-{region.lower()}-vol-{idx}",
        "type": "block_volume",
        "region": region,
        "environment": env,
        "team": team,
        "size_gb": random.choice([100, 200, 500, 1000]),
        "monthly_cost_inr": random.choice([300, 450, 680, 1100]),
        "status": "orphaned" if random.random() < 0.25 else "attached",
    }


def generate_public_ip(region: str, idx: int, env: str) -> Dict[str, Any]:
    return {
        "id": f"ip-{region.lower()}-{idx}",
        "name": f"{env}-{region.lower()}-ip-{idx}",
        "type": "public_ip",
        "region": region,
        "environment": env,
        "team": "shared",
        "monthly_cost_inr": random.choice([200, 350, 500]),
        "status": "orphaned" if random.random() < 0.3 else "attached",
    }


def generate_vm_metrics(vm: Dict[str, Any]) -> List[Dict[str, Any]]:
    region_factor = {"Noida": 1.0, "Bangalore": 0.95, "Mumbai": 1.08}[vm["region"]]
    team_factor = {"core-backend": 1.2, "analytics": 1.28, "frontend": 0.92, "qa": 0.85}[vm["team"]]
    env_factor = {"prod": 1.2, "staging": 0.82, "dev": 0.7}[vm["environment"]]
    base_cpu = 36 * region_factor * team_factor * env_factor
    base_mem = 40 * region_factor * team_factor * env_factor

    series: List[Dict[str, Any]] = []
    start = datetime(2026, 10, 1, 0, 0)

    for day in range(30):
        for hour in range(24):
            for minute in [0, 15, 30, 45]:
                ts = start + timedelta(days=day, hours=hour, minutes=minute)
                day_of_week = ts.weekday()

                workload = 1.0
                if 2 <= hour <= 8:
                    workload = 0.45
                elif 9 <= hour <= 17:
                    workload = 1.3
                elif 18 <= hour <= 22:
                    workload = 1.1

                if vm["environment"] in ["staging", "dev"] and day_of_week >= 5:
                    workload *= 0.45

                sudden_spike = (
                    vm["environment"] == "prod"
                    and (day in [6, 12, 19, 24])
                    and hour in [5, 6, 7]
                    and minute in [0, 15, 30]
                )

                cpu_base = base_cpu * workload
                mem_base = base_mem * workload

                if sudden_spike:
                    cpu_base *= 1.8
                    mem_base *= 1.7

                cpu = clamp(cpu_base + np.random.normal(0, 10), 0, 100)
                mem = clamp(mem_base + np.random.normal(0, 8), 0, 100)
                iops = max(0, 800 + (cpu * 35) + np.random.normal(0, 90))

                series.append(
                    {
                        "timestamp": ts.isoformat(),
                        "cpu_percent": round(float(cpu), 2),
                        "memory_percent": round(float(mem), 2),
                        "iops": round(float(iops), 2),
                    }
                )

    return series


def build_dataset() -> Dict[str, Any]:
    inventory: List[Dict[str, Any]] = []
    metrics: Dict[str, List[Dict[str, Any]]] = {}

    vm_count = 0
    for region in REGIONS:
        for env in ENVIRONMENTS:
            for team in TEAMS:
                for i in range(8):
                    vm = generate_vm_resource(region, vm_count + 1, env, team)
                    inventory.append(vm)
                    metrics[vm["id"]] = generate_vm_metrics(vm)
                    vm_count += 1

    for region in REGIONS:
        for env in ENVIRONMENTS:
            for team in TEAMS:
                for i in range(3):
                    inventory.append(generate_block_volume(region, i + 1, env, team))
                for i in range(2):
                    inventory.append(generate_public_ip(region, i + 1, env))

    return {"inventory": inventory, "metrics": metrics}


DATASET = build_dataset()


def compute_p95(arr: List[float]) -> float:
    if not arr:
        return 0.0
    sorted_arr = sorted(arr)
    idx = max(0, min(len(sorted_arr) - 1, int(len(sorted_arr) * 0.95)))
    return sorted_arr[idx]


def compute_p99(arr: List[float]) -> float:
    if not arr:
        return 0.0
    sorted_arr = sorted(arr)
    idx = max(0, min(len(sorted_arr) - 1, int(len(sorted_arr) * 0.99)))
    return sorted_arr[idx]


def summarize_vm(resource: Dict[str, Any]) -> Dict[str, Any]:
    series = DATASET["metrics"].get(resource["id"], [])
    if not series:
        return {
            "cpu_p95": 0,
            "cpu_p99": 0,
            "mem_p95": 0,
            "mem_p99": 0,
            "avg_cpu": 0,
            "avg_memory": 0,
            "peak_cpu": 0,
            "peak_mem": 0,
            "iops_peak": 0,
        }

    cpu_vals = [x["cpu_percent"] for x in series]
    mem_vals = [x["memory_percent"] for x in series]
    iops_vals = [x["iops"] for x in series]

    return {
        "cpu_p95": round(float(compute_p95(cpu_vals)), 2),
        "cpu_p99": round(float(compute_p99(cpu_vals)), 2),
        "mem_p95": round(float(compute_p95(mem_vals)), 2),
        "mem_p99": round(float(compute_p99(mem_vals)), 2),
        "avg_cpu": round(float(np.mean(cpu_vals)), 2),
        "avg_memory": round(float(np.mean(mem_vals)), 2),
        "peak_cpu": round(float(max(cpu_vals)), 2),
        "peak_mem": round(float(max(mem_vals)), 2),
        "iops_peak": round(float(max(iops_vals)), 2),
    }


def list_recommendations() -> List[Dict[str, Any]]:
    actions: List[Dict[str, Any]] = []

    for resource in DATASET["inventory"]:
        if resource["type"] == "vm":
            summary = summarize_vm(resource)
            current_tier = next((t for t in VM_TIERS if t["id"] == resource["tier"]), VM_TIERS[0])

            if summary["cpu_p99"] < 40 and summary["mem_p99"] < 50:
                target_tier = downsize_tier(current_tier)
                actions.append(
                    {
                        "id": f"{resource['id']}-downsize",
                        "resource_id": resource["id"],
                        "resource_name": resource["name"],
                        "category": "DOWNSIZE",
                        "action_type": "DOWNSIZE",
                        "region": resource["region"],
                        "environment": resource["environment"],
                        "team": resource["team"],
                        "current_tier": resource["tier"],
                        "recommended_tier": target_tier["id"],
                        "savings_inr": round(resource["monthly_cost_inr"] - target_tier["monthly_inr"], 2),
                        "risk": "LOW",
                        "approved": True,
                        "reason": "p99 CPU and memory remain comfortably below the safety threshold.",
                        "projected_cost_inr": target_tier["monthly_inr"],
                    }
                )

            if summary["cpu_p95"] > 85 or summary["mem_p95"] > 85:
                target_tier = upsize_tier(current_tier)
                actions.append(
                    {
                        "id": f"{resource['id']}-upsize",
                        "resource_id": resource["id"],
                        "resource_name": resource["name"],
                        "category": "UPSIZE (PERFORMANCE RISK)",
                        "action_type": "UPSIZE",
                        "region": resource["region"],
                        "environment": resource["environment"],
                        "team": resource["team"],
                        "current_tier": resource["tier"],
                        "recommended_tier": target_tier["id"],
                        "savings_inr": 0.0,
                        "risk": "MEDIUM",
                        "approved": True,
                        "reason": "p95 CPU/RAM exceed safer workload thresholds, so vertical scale-out reduces crash risk.",
                        "projected_cost_inr": target_tier["monthly_inr"],
                        "investment_inr": target_tier["monthly_inr"] - resource["monthly_cost_inr"],
                    }
                )

            if resource["environment"] in ["staging", "dev"]:
                actions.append(
                    {
                        "id": f"{resource['id']}-schedule",
                        "resource_id": resource["id"],
                        "resource_name": resource["name"],
                        "category": "SCHEDULE (CRON)",
                        "action_type": "SCHEDULE",
                        "region": resource["region"],
                        "environment": resource["environment"],
                        "team": resource["team"],
                        "current_tier": resource["tier"],
                        "recommended_tier": "micro",
                        "savings_inr": round(resource["monthly_cost_inr"] * 0.3, 2),
                        "risk": "LOW",
                        "approved": True,
                        "reason": "Weekend and early morning auto-stop policy reduces wasted spend during idle windows.",
                        "projected_cost_inr": resource["monthly_cost_inr"] * 0.7,
                    }
                )

        elif resource["type"] in ["block_volume", "public_ip"]:
            if resource["status"] == "orphaned":
                actions.append(
                    {
                        "id": f"{resource['id']}-terminate",
                        "resource_id": resource["id"],
                        "resource_name": resource["name"],
                        "category": "TERMINATE ORPHAN",
                        "action_type": "TERMINATE",
                        "region": resource["region"],
                        "environment": resource["environment"],
                        "team": resource["team"],
                        "current_tier": resource["type"],
                        "recommended_tier": "terminate",
                        "savings_inr": float(resource.get("monthly_cost_inr", 0)),
                        "risk": "LOW",
                        "approved": True,
                        "reason": "Detached asset has zero utilization and can be removed immediately.",
                        "projected_cost_inr": 0,
                    }
                )

    return actions


def get_summary() -> Dict[str, Any]:
    vm_total = sum(r.get("monthly_cost_inr", 0) for r in DATASET["inventory"] if r["type"] == "vm")
    orphan_total = sum(
        r.get("monthly_cost_inr", 0)
        for r in DATASET["inventory"]
        if r["type"] in ["block_volume", "public_ip"] and r["status"] == "orphaned"
    )
    recommendations = list_recommendations()
    approved_savings = sum(a.get("savings_inr", 0) for a in recommendations if a.get("approved"))

    return {
        "total_monthly_spend_inr": vm_total + orphan_total,
        "projected_optimized_spend_inr": round(vm_total - approved_savings, 2),
        "monthly_savings_inr": round(approved_savings, 2),
        "reliability_health_index": 87,
        "orphan_cost_inr": orphan_total,
        "resource_count": len(DATASET["inventory"]),
        "vm_count": sum(1 for x in DATASET["inventory"] if x["type"] == "vm"),
        "data_source_note": "Public Azure VM pricing benchmark + synthetic workload simulation",
    }


@app.get("/api/resources")
def get_resources():
    return {
        "inventory": DATASET["inventory"],
        "summary": get_summary(),
    }


@app.get("/api/metrics/{resource_id}")
def get_metrics(resource_id: str):
    resource = next((r for r in DATASET["inventory"] if r["id"] == resource_id), None)
    if not resource:
        return {"error": "resource not found"}
    return {
        "resource": resource,
        "metrics": DATASET["metrics"].get(resource_id, []),
        "summary": summarize_vm(resource),
    }


@app.get("/api/recommendations")
def get_recommendations():
    return {
        "actions": list_recommendations(),
        "summary": get_summary(),
    }


class SimulationRequest(BaseModel):
    early_window_start: int = 2
    early_window_end: int = 9
    peak_threshold: float = 75
    cool_down_minutes: int = 15

            "end_hour": payload.early_window_end,
            "peak_threshold": payload.peak_threshold,
            "cool_down_minutes": payload.cool_down_minutes,

        "simulation_log": [
            {"time": "2026-10-01 02:00", "event": "CRON SCALING", "detail": "Production VM tier reduced during early morning low-load window."},
